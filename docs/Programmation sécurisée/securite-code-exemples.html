  --name sonarqube \

  -e SONAR_JDBC_PASSWORD=sonarpassword \

Exemples courants et corrections rapides (Quick Fixes)

❌ Pattern SQL Injection

❌ Code vulnérable (détecté par Rider)
string userId = GetUserId();
string query = $"SELECT * FROM users WHERE id = {userId}";
var result = db.ExecuteQuery(query);
✅ Code sécurisé
string userId = GetUserId();
using (SqlCommand cmd = new SqlCommand(
    "SELECT * FROM users WHERE id = @id", connection))
{
    cmd.Parameters.AddWithValue("@id", userId);
    var result = cmd.ExecuteReader();
}

❌ Pattern Weak Cryptography

❌ Algorithme faible (détecté)
using (MD5 md5 = MD5.Create())
{
    byte[] hash = md5.ComputeHash(
        Encoding.UTF8.GetBytes(password));
}
✅ Algorithme fort
// Option 1: SHA256
using (SHA256 sha256 = SHA256.Create())
{
    byte[] hash = sha256.ComputeHash(
        Encoding.UTF8.GetBytes(password));
}

// Option 2: Bcrypt (recommandé pour mots de passe)
string hashedPassword = BCrypt.Net.BCrypt
    .HashPassword(password);

❌ Pattern Hardcoded Secret

❌ Secret en dur (détecté)
private const string ApiKey = 
    "sk-1234567890abcdef";
private const string DbPassword = 
    "SuperSecurePassword123";
✅ Depuis configuration
// appsettings.json
{
  "Secrets": {
    "ApiKey": "value_from_env",
    "DbPassword": "value_from_env"
  }
}

// Code
var config = new ConfigurationBuilder()
    .AddUserSecrets()
    .Build();
string apiKey = config["Secrets:ApiKey"];

📋 Checklist de sécurité finale

  • ✓ Code SAST : SonarQube et Rider configurés avec OWASP/CWE mapping
  • ✓ Dépendances : Snyk et npm audit intégrés en CI/CD
  • ✓ Secrets : Scanning avec TruffleHog en pré-commit
  • ✓ Quality Gates : Zéro vulnérabilité CRITIQUE et 0 hotspot
  • ✓ Logging : Aucune donnée sensible dans les logs
